\[
\therefore \frac{4 (x^2 + 1)}{3} = B (x^2 + 1), \quad \therefore B = \frac{4}{3}
\]
\[
\int \frac{x^2 dx}{(x^2 + 1)(x^2 + 4)} = -\frac{1}{3} \int \left\{ \frac{dx}{(x^2 + 1)} - \int \frac{4 dx}{(x^2 + 4)} \right\}
\]
\[
= \frac{1}{3} \int \left\{ \frac{dx}{(1 + x^2)} - \frac{dx}{\left(1 + \frac{x^2}{4}\right)} \right\}
\]
\[
= \frac{1}{3} \left\{ 2 \tan^{-1} \frac{x}{2} - \tan^{-1} x \right\}.
\]
